Data Cleaning¶

In [4]:
FULL = pd.date_range(train.date.min(), test.date.max())

# 1. Oil: fill the missing prices by interpolation
oil_f = oil.set_index("date").reindex(FULL).rename(columns={"dcoilwtico": "oil"})
oil_f["oil"] = oil_f["oil"].interpolate(limit_direction="both")
oil_f = oil_f.rename_axis("date").reset_index()

# 2. Stores not yet open: drop rows before a store's first sale
sd = train.groupby(["store_nbr", "date"], as_index=False).sales.sum()
open_date = sd[sd.sales > 0].groupby("store_nbr").date.min().rename("open_date")
train_c = train.merge(open_date, on="store_nbr")
train_c = train_c[train_c.date >= train_c.open_date].drop(columns="open_date")
print("Rows removed (store not open yet):", len(train) - len(train_c))   # 222,057

# 3. Holidays: keep only days that were really celebrated
h = holidays[(~holidays.transferred) & (holidays.type != "Work Day")]
events = set(h.loc[h.type == "Event", "date"]); h = h[h.type != "Event"]
nat = set(h.loc[h.locale == "National", "date"])
reg = set(zip(h.loc[h.locale == "Regional", "date"], h.loc[h.locale == "Regional", "locale_name"]))
loc = set(zip(h.loc[h.locale == "Local", "date"],    h.loc[h.locale == "Local", "locale_name"]))

# 4. Calendar and store-by-date table
cal = pd.DataFrame({"date": FULL})
cal["dow"], cal["dom"], cal["month"] = cal.date.dt.dayofweek, cal.date.dt.day, cal.date.dt.month
cal["doy"] = cal.date.dt.dayofyear
cal["weekofyear"] = cal.date.dt.isocalendar().week.astype(int)
cal["payday"] = ((cal.dom == 15) | cal.date.dt.is_month_end).astype(int)   # wages paid 15th and month-end
cal["hol_national"] = cal.date.isin(nat).astype(int)
cal["event"] = cal.date.isin(events).astype(int)
cal["quake_period"] = ((cal.date >= "2016-04-16") & (cal.date <= "2016-05-16")).astype(int)
cal = cal.merge(oil_f, on="date")
cal["oil_ma7"] = cal.oil.rolling(7, min_periods=1).mean()

sc = stores.merge(cal, how="cross")
sc["hol_regional"] = [int((d, s) in reg) for d, s in zip(sc.date, sc.state)]
sc["hol_local"]    = [int((d, c) in loc) for d, c in zip(sc.date, sc.city)]

Several cleaning procedures are performed:

Missing oil prices are filled using interpolation. Oil column is renamed for easier handling. Store opening dates are estimated using first positive sales records. Records before store opening dates are removed. Holiday and event data are cleaned and categorized. Calendar features are created.

Data cleaning improves model quality by removing noise and ensuring consistency.

Important findings:

43 missing oil price values were successfully repaired. Pre-opening store records were removed because they do not represent actual business activity. Holiday information helps capture special demand patterns.

KEYS = ["store_nbr", "family"]; TRAIN_END = train.date.max()
S = train.pivot_table(index="date", columns=KEYS, values="sales", aggfunc="sum").reindex(FULL)
P = (pd.concat([train, test]).pivot_table(index="date", columns=KEYS, values="onpromotion", aggfunc="sum")
       .reindex(index=FULL, columns=S.columns).fillna(0))
S.loc[:TRAIN_END] = S.loc[:TRAIN_END].fillna(0)      # 25 Dec = closed = 0 sales
LS = np.log1p(S); base = LS.shift(16)

feat = {f"lag_{L}": LS.shift(L) for L in (16, 21, 28, 35, 364)}
feat.update({f"roll_mean_{w}": base.rolling(w, min_periods=1).mean() for w in (7, 14, 28, 56)})
feat["roll_std_28"] = base.rolling(28, min_periods=2).std()
feat["wd_mean_4"] = (LS.shift(21) + LS.shift(28) + LS.shift(35) + LS.shift(42)) / 4
feat["promo"], feat["promo_roll7"] = P, P.rolling(7, min_periods=1).mean()
feat["target"] = LS

T, C = LS.shape
df = pd.DataFrame({k: v.values.ravel().astype("float32") for k, v in feat.items()})

def make_ridge():
    onehot = ["store_nbr", "family_code", "type_code", "city_code", "state_code", "dow", "month"]
    numeric = [c for c in FEATURES if c not in onehot]
    pre = ColumnTransformer([("cat", OneHotEncoder(handle_unknown="ignore"), onehot),
                             ("num", StandardScaler(), numeric)])
    return Pipeline([("prep", pre), ("model", Ridge(alpha=1.0))])

def evaluate(y_log, p_log):
    yt, yp = np.expm1(y_log), np.clip(np.expm1(p_log), 0, None)
    wape = np.abs(yt - yp).sum() / yt.sum()
    return {"RMSLE": np.sqrt(np.mean((np.log1p(yp) - y_log) ** 2)),
            "MAE": np.abs(yt - yp).mean(),
            "R2": 1 - ((yt - yp) ** 2).sum() / ((yt - yt.mean()) ** 2).sum(),
            "Accuracy_%": 100 * (1 - wape)}

results = {"Baseline (same weekday, 4-wk avg)": evaluate(y_va, va.wd_mean_4.values)}
